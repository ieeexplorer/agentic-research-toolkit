   178→// ─── Helpers ──────────────────────────────────────────────────────────────────
   179→
   180→function timeAgo(d: string) {
   181→  const mins = Math.floor((Date.now() - new Date(d).getTime()) / 60000);
   182→  if (mins < 1) return "just now";
   183→  if (mins < 60) return `${mins}m ago`;
   184→  const h = Math.floor(mins / 60);
   185→  if (h < 24) return `${h}h ago`;
   186→  return `${Math.floor(h / 24)}d ago`;
   187→}
   188→
   189→function formatDuration(ms: number): string {
   190→  if (ms < 1000) return `${ms}ms`;
   191→  return `${(ms / 1000).toFixed(1)}s`;
   192→}
   193→
   194→function parseNodeResults(raw: string): NodeResult[] {
   195→  try {
   196→    const parsed = JSON.parse(raw);
   197→    if (Array.isArray(parsed)) return parsed;
   198→    return [];
   199→  } catch {
   200→    return [];
   201→  }
   202→}

   262→    }

   269→      const res = await fetch("/api/agents");
   270→      if (res.ok) setAgents(await res.json());

   272→      /* ignore */

   306→        method: "POST",

   312→      setNewWf({ name: "", description: "" });

   315→      toast.error("Failed to create workflow");
   316→    }

   327→    }

   338→      fetch(`/api/workflows/${selectedWorkflowId}`)

   341→      fetchWorkflows();

   343→      toast.error("Failed to delete node");
   344→    }

   351→      const res = await fetch(

   357→        }
   358→      );
   359→      const exec = await res.json();
   360→      toast.success(`Workflow completed in ${formatDuration(exec.duration)}`);
   361→      setExecInput("");
   362→      fetch(`/api/workflows/${selectedWorkflowId}`)
   363→        .then((r) => r.json())
   364→        .then(setSelected);
   365→      fetchWorkflows();
   366→    } catch {
   367→      toast.error("Execution failed");
   368→    }
   369→    setExecuting(false);
   370→  };
   371→
   372→  const handleStatusToggle = async () => {
   373→    if (!selected) return;
   374→    const cycle = STATUS_CYCLE[selected.status];
   375→    if (!cycle) return;
   376→    setTogglingStatus(true);
   377→    try {
   378→      await fetch(`/api/workflows/${selected.id}`, {
   379→        method: "PUT",
   380→        headers: { "Content-Type": "application/json" },
   381→        body: JSON.stringify({ status: cycle.next }),
   382→      });
   383→      toast.success(`Status changed to ${cycle.next}`);
   384→      fetchWorkflows();
   385→      fetch(`/api/workflows/${selected.id}`)
   386→        .then((r) => r.json())
   387→        .then(setSelected);
   388→    } catch {
   389→      toast.error("Failed to update status");
   390→    }
   391→    setTogglingStatus(false);
   392→  };
   393→

   397→    try {
   398→      // Auto-position: right of existing nodes

   411→        nodeType: newNodeType,

   416→      if (newNodeType === "agent" && newNodeAgentId) {

   418→      }
   419→      if (newNodeType === "tool" && newNodeToolId) {
   420→        body.toolId = newNodeToolId;
   421→      }

   423→      await fetch(`/api/workflows/${selectedWorkflowId}/nodes`, {
   424→        method: "POST",
   425→        headers: { "Content-Type": "application/json" },

   486→              markerWidth="10"
   487→              markerHeight="7"

   489→              refY="3.5"

   492→              <polygon
   493→                points="0 0, 10 3.5, 0 7"

   502→                key={`shadow-${type}`}

   504→                x="-20%"
   505→                y="-20%"
   506→                width="140%"
   507→                height="140%"
   508→              >
   509→                <feDropShadow
   510→                  dx="0"
   511→                  dy="2"
   512→                  stdDeviation="3"
   513→                  floodOpacity="0.08"
   514→                />
   515→              </filter>
   516→            ))}
   517→            <filter id="shadow-hover" x="-20%" y="-20%" width="140%" height="140%">
   518→              <feDropShadow
   519→                dx="0"
   520→                dy="4"
   521→                stdDeviation="6"
   522→                floodOpacity="0.18"
   523→              />
   524→            </filter>

   526→
   527→          <style>{`
   528→            @keyframes dash-flow {
   529→              to { stroke-dashoffset: -16; }

   536→          {/* Connections */}

   538→            const src = nodeMap.get(conn.sourceNodeId);
   539→            const tgt = nodeMap.get(conn.targetNodeId);
   540→            if (!src || !tgt) return null;
   541→            const x1 = src.positionX + nodeW + padX;
   542→            const y1 = src.positionY + nodeH / 2 + padY;
   543→            const x2 = tgt.positionX + padX;
   544→            const y2 = tgt.positionY + nodeH / 2 + padY;
   545→            const cx = (x1 + x2) / 2;
   546→
   547→            return (
   548→              <g key={conn.id}>
   549→                {/* Background glow for the path */}
   550→                <path
   551→                  d={`M ${x1} ${y1} C ${cx} ${y1}, ${cx} ${y2}, ${x2} ${y2}`}
   552→                  fill="none"
   553→                  className="stroke-muted-foreground/10"

   560→                  fill="none"
   561→                  className="stroke-muted-foreground/50 connection-line"
   562→                  strokeWidth="2"
   563→                  strokeDasharray="8 8"
   564→                  strokeLinecap="round"
   565→                  markerEnd="url(#arrowhead-animated)"
   566→                />

   568→                  <text
   569→                    x={cx}
   570→                    y={Math.min(y1, y2) - 8}
   571→                    textAnchor="middle"
   572→                    className="fill-muted-foreground text-[10px] font-mono"

   590→              node.agent?.name || node.tool?.name || "";

   592→            return (

   594→                <Tooltip>
   595→                  <TooltipTrigger asChild>
   596→                    <g
   597→                      onMouseEnter={() => setHoveredNodeId(node.id)}
   598→                      onMouseLeave={() => setHoveredNodeId(null)}
   599→                      style={{

   604→                      }}
