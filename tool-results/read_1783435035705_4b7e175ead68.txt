    64→function timeAgo(d: string) {
    65→  const mins = Math.floor((Date.now() - new Date(d).getTime()) / 60000);
    66→  if (mins < 1) return "just now";
    67→  if (mins < 60) return `${mins}m ago`;
    68→  const h = Math.floor(mins / 60);
    69→  if (h < 24) return `${h}h ago`;
    70→  return `${Math.floor(h / 24)}d ago`;
    71→}

   114→      // Use setTimeout to avoid synchronous setState in effect

   116→      return () => clearTimeout(id);

   122→      method: "POST",

   130→  };

   144→        method: "PUT",
   145→        headers: { "Content-Type": "application/json" },
   146→        body: JSON.stringify(editTool),
   147→      });
   148→      toast.success("Tool updated");

   150→      fetchTools();
   151→      fetch(`/api/tools/${selectedId}`).then((r) => r.json()).then(setSelected);
   152→    } catch {
   153→      toast.error("Failed to update tool");
   154→    }
   155→    setUpdating(false);
   156→  };
   157→
   158→  const openEditDialog = () => {
   159→    if (!selected) return;
   160→    setEditTool({
   161→      name: selected.name,
   162→      description: selected.description,
   163→      endpoint: selected.endpoint,

   167→  };

   170→    if (!selectedId) return;
   171→    await fetch(`/api/tools/${selectedId}`, {
   172→      method: "PUT",
   173→      headers: { "Content-Type": "application/json" },
   174→      body: JSON.stringify({ status: newStatus }),
   175→    });
   176→    toast.success(`Status changed to ${newStatus}`);
   177→    fetchTools();
   178→    fetch(`/api/tools/${selectedId}`).then((r) => r.json()).then(setSelected);
   179→  };
   180→
   181→  const handleExecute = async () => {
   182→    if (!selectedId) return;

   185→      const res = await fetch(`/api/tools/${selectedId}/execute`, {

   187→        headers: { "Content-Type": "application/json" },

   300→                                  <p className="font-medium text-sm truncate">{tool.name}</p>

   308→                                    <TooltipTrigger asChild>
   309→                                      <Badge variant="outline" className="text-[10px] px-1.5 py-0 gap-1">
   310→                                        <Workflow className="h-2.5 w-2.5" />{tool._count.workflowNodes}
   311→                                      </Badge>
   312→                                    </TooltipTrigger>
   313→                                    <TooltipContent>{tool._count.workflowNodes} workflow{tool._count.workflowNodes > 1 ? "s" : ""}</TooltipContent>
   314→                                  </Tooltip>
   315→                                ) : null}
   316→                                <ChevronRight className={`h-4 w-4 text-muted-foreground transition-all duration-200 ${selectedId === tool.id ? "opacity-100 translate-x-0" : "opacity-0 group-hover:opacity-100 -translate-x-1 group-hover:translate-x-0"}`} />
   317→                              </div>
   318→                            </div>

   323→                  </div>

   326→            </CardContent>

   332→          <AnimatePresence mode="wait">
   333→            {selected ? (() => {

   335→              const sc = STATUS_CONFIG[selected.status] || STATUS_CONFIG.draft;
   336→              const Icon = tc.icon;

   343→                  transition={{ duration: 0.25 }}
   344→                >
   345→                  <Card>
   346→                    <CardHeader>
   347→                      <div className="flex items-start justify-between gap-4">
   348→                        <div className="flex items-center gap-3 min-w-0">
   349→                          <div className={`p-3 rounded-xl ${tc.bg} flex-shrink-0`}>
   350→                            <Icon className={`h-6 w-6 ${tc.color}`} />
   351→                          </div>
   352→                          <div className="min-w-0">
   353→                            <CardTitle className="flex items-center gap-2 flex-wrap">

   447→                      {/* Execution History */}

   527→          </AnimatePresence>
   528→        </div>
   529→      </div>
   530→
   531→      {/* Edit Dialog */}
   532→      <Dialog open={editOpen} onOpenChange={setEditOpen}>
