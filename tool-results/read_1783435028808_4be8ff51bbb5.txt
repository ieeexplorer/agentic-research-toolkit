   114→function timeAgo(d: string) {
   115→  const mins = Math.floor((Date.now() - new Date(d).getTime()) / 60000);
   116→  if (mins < 1) return "just now";
   117→  if (mins < 60) return `${mins}m ago`;
   118→  const h = Math.floor(mins / 60);
   119→  if (h < 24) return `${h}h ago`;
   120→  return `${Math.floor(h / 24)}d ago`;
   121→}

   123→function TokenUsageBar({ tokensUsed }: { tokensUsed: number }) {
   124→  const maxRef = 8000;
   125→  const pct = Math.min((tokensUsed / maxRef) * 100, 100);
   126→  const color =
   127→    pct < 30
   128→      ? "text-emerald-600 dark:text-emerald-400"
   129→      : pct < 70
   130→        ? "text-amber-600 dark:text-amber-400"
   131→        : "text-rose-600 dark:text-rose-400";
   132→  return (
   133→    <div className="flex items-center gap-2">
   134→      <div className="h-1.5 w-16 rounded-full bg-muted overflow-hidden">
   135→        <motion.div
   136→          className={`h-full rounded-full ${pct < 30 ? "bg-emerald-500" : pct < 70 ? "bg-amber-500" : "bg-rose-500"}`}
   137→          initial={{ width: 0 }}
   138→          animate={{ width: `${pct}%` }}
   139→          transition={{ duration: 0.5, ease: "easeOut" }}
   140→        />
   141→      </div>
   142→      <span className={`text-xs font-medium ${color}`}>{tokensUsed.toLocaleString()}</span>
   143→    </div>
   144→  );
   145→}

   315→export function AgentsView() {
   316→  const { selectedAgentId, setSelectedAgentId, refreshing } = useToolkitStore();
   317→  const [agents, setAgents] = useState<Agent[]>([]);
   318→  const [detailAgent, setDetailAgent] = useState<Agent | null>(null);
   319→  const [loading, setLoading] = useState(true);
   320→  const [executing, setExecuting] = useState(false);
   321→  const [createOpen, setCreateOpen] = useState(false);
   322→  const [editOpen, setEditOpen] = useState(false);
   323→  const [editAgent, setEditAgent] = useState({
   324→    name: "",
   325→    description: "",
   326→    type: "research",
   327→    systemPrompt: "",
   328→    model: "gpt-4o",
   329→    temperature: 0.7,
   330→  });
   331→  const [editSaving, setEditSaving] = useState(false);
   332→  const [newAgent, setNewAgent] = useState({
   333→    name: "",
   334→    description: "",
   335→    type: "research",
   336→    systemPrompt: "",
   337→    model: "gpt-4o",
   338→    temperature: 0.7,
   339→  });
   340→  const [execInput, setExecInput] = useState("");
   341→
   342→  const fetchAgents = useCallback(async () => {
   343→    setLoading(true);
   344→    try {
   345→      const res = await fetch("/api/agents");
   346→      const data = await res.json();
   347→      setAgents(data);
   348→    } finally {
   349→      setLoading(false);
   350→    }
   351→  }, []);
   352→
   353→  useEffect(() => {
   354→    fetchAgents();
   355→  }, [fetchAgents, refreshing]);
   356→
   357→  useEffect(() => {
   358→    if (!selectedAgentId) {
   359→      setDetailAgent(null);
   360→      return;
   361→    }
   362→    let cancelled = false;
   363→    fetch(`/api/agents/${selectedAgentId}`)
   364→      .then((r) => r.json())
   365→      .then((data) => {
   366→        if (!cancelled) setDetailAgent(data);
   367→      })
   368→      .catch(() => {});
   369→    return () => {
   370→      cancelled = true;
   371→    };
   372→  }, [selectedAgentId, refreshing]);
   373→
   374→  const handleCreate = async () => {
   375→    await fetch("/api/agents", {
   376→      method: "POST",
   377→      headers: { "Content-Type": "application/json" },
   378→      body: JSON.stringify(newAgent),
   379→    });
   380→    toast.success("Agent created");
   381→    setCreateOpen(false);
   382→    setNewAgent({ name: "", description: "", type: "research", systemPrompt: "", model: "gpt-4o", temperature: 0.7 });
   383→    fetchAgents();
   384→  };
   385→
   386→  const handleDelete = async (id: string) => {
   387→    await fetch(`/api/agents/${id}`, { method: "DELETE" });
   388→    toast.success("Agent deleted");
   389→    if (selectedAgentId === id) setSelectedAgentId(null);
   390→    fetchAgents();
   391→  };
   392→
   393→  const handleExecute = async () => {
   394→    if (!selectedAgentId) return;
   395→    setExecuting(true);
   396→    try {
   397→      const res = await fetch(`/api/agents/${selectedAgentId}/execute`, {
   398→        method: "POST",
   399→        headers: { "Content-Type": "application/json" },
   400→        body: JSON.stringify({ input: execInput || "Run analysis" }),
   401→      });
   402→      const execution = await res.json();
   403→      toast.success(`Execution completed in ${execution.duration}ms`);
   404→      setExecInput("");
   405→      fetch(`/api/agents/${selectedAgentId}`)
   406→        .then((r) => r.json())
   407→        .then(setDetailAgent);
   408→      fetchAgents();
   409→    } catch {
   410→      toast.error("Execution failed");
   411→    }
   412→    setExecuting(false);
   413→  };
   414→
   415→  const openEditDialog = () => {
   416→    if (!detailAgent) return;
   417→    setEditAgent({
   418→      name: detailAgent.name,
   419→      description: detailAgent.description,
   420→      type: detailAgent.type,
   421→      systemPrompt: detailAgent.systemPrompt,
   422→      model: detailAgent.model,
   423→      temperature: detailAgent.temperature,
   424→    });
   425→    setEditOpen(true);
   426→  };
   427→
   428→  const handleEditSave = async () => {
   429→    if (!detailAgent) return;
   430→    setEditSaving(true);
   431→    try {
   432→      const res = await fetch(`/api/agents/${detailAgent.id}`, {
   433→        method: "PUT",
   434→        headers: { "Content-Type": "application/json" },
   435→        body: JSON.stringify(editAgent),
   436→      });
   437→      if (!res.ok) throw new Error("Failed to save");
   438→      toast.success("Agent updated");
   439→      setEditOpen(false);
   440→      // Refresh detail
   441→      fetch(`/api/agents/${detailAgent.id}`)
   442→        .then((r) => r.json())
   443→        .then(setDetailAgent);
   444→      fetchAgents();
   445→    } catch {
   446→      toast.error("Failed to update agent");
   447→    }
   448→    setEditSaving(false);
   449→  };
   450→
   451→  const handleStatusToggle = async (newStatus: string) => {
   452→    if (!detailAgent || detailAgent.status === newStatus) return;
   453→    try {
   454→      const res = await fetch(`/api/agents/${detailAgent.id}`, {
   455→        method: "PUT",
   456→        headers: { "Content-Type": "application/json" },
   457→        body: JSON.stringify({ status: newStatus }),
   458→      });
   459→      if (!res.ok) throw new Error();
   460→      toast.success(`Status changed to ${newStatus}`);
   461→      fetch(`/api/agents/${detailAgent.id}`)
   462→        .then((r) => r.json())
   463→        .then(setDetailAgent);
   464→      fetchAgents();
   465→    } catch {
   466→      toast.error("Failed to change status");
   467→    }
   468→  };
   469→
   470→  return (
   471→    <div className="space-y-6">
   472→      {/* Header */}
   473→      <div className="flex items-center justify-between">
   474→        <motion.div initial={{ opacity: 0, y: -8 }} animate={{ opacity: 1, y: 0 }}>
   475→          <h2 className="text-2xl font-bold tracking-tight">Agents</h2>
   476→          <p className="text-muted-foreground text-sm mt-0.5">
   477→            Create and manage your research agents.
   478→          </p>
   479→        </motion.div>
   480→        <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   481→          <DialogTrigger asChild>
   482→            <Button>
   483→              <Plus className="h-4 w-4 mr-2" />
   484→              New Agent
   485→            </Button>
   486→          </DialogTrigger>
   487→          <DialogContent className="sm:max-w-lg max-h-[90vh] overflow-y-auto">
   488→            <DialogHeader>
   489→              <DialogTitle>Create Agent</DialogTitle>
   490→            </DialogHeader>
   491→            <div className="space-y-4 pt-2">
   492→              <div className="grid grid-cols-2 gap-4">
   493→                <div className="space-y-2">
   494→                  <Label>Name</Label>
   495→                  <Input
   496→                    value={newAgent.name}
   497→                    onChange={(e) => setNewAgent({ ...newAgent, name: e.target.value })}
   498→                    placeholder="Research Analyst"
   499→                  />
   500→                </div>
   501→                <div className="space-y-2">
   502→                  <Label>Type</Label>
   503→                  <Select
   504→                    value={newAgent.type}
   505→                    onValueChange={(v) => setNewAgent({ ...newAgent, type: v })}
   506→                  >
   507→                    <SelectTrigger>
   508→                      <SelectValue />
   509→                    </SelectTrigger>
   510→                    <SelectContent>
   511→                      {TYPE_OPTIONS.map((t) => (
   512→                        <SelectItem key={t} value={t}>
   513→                          {TYPE_ICONS[t]} {t.charAt(0).toUpperCase() + t.slice(1)}
   514→                        </SelectItem>
   515→                      ))}
   516→                    </SelectContent>
   517→                  </Select>
   518→                </div>
   519→              </div>
   520→              <div className="space-y-2">
   521→                <Label>Description</Label>
   522→                <Textarea
   523→                  value={newAgent.description}
   524→                  onChange={(e) => setNewAgent({ ...newAgent, description: e.target.value })}
   525→                  placeholder="What does this agent do?"
   526→                  rows={2}
   527→                />
   528→              </div>
   529→              <div className="space-y-2">
   530→                <Label>System Prompt</Label>
   531→                <Textarea
   532→                  value={newAgent.systemPrompt}
   533→                  onChange={(e) => setNewAgent({ ...newAgent, systemPrompt: e.target.value })}

   543→                    onValueChange={(v) => setNewAgent({ ...newAgent, model: v })}

   545→                    <SelectTrigger>
   546→                      <SelectValue />
